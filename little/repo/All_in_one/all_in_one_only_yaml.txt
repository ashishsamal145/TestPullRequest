# deployScorejobmaifests.yaml

# job variables from pipeline
#        $(jobName)
#        $(jobManifest)
#        input build pipelines

parameters:
  DEPLOYMENT_SERVICE_CONNECTION: 'NOT_DEFINE'
  NAMESPACE: 'NOT_DEFINE'
  AZURE_STORAGE_ACCOUNT: 'NOT_DEFINE'
  AZURE_STORAGE_ACCOUNT_SAS_TOKEN: 'NOT_DEFINE'
  ARTIFACT_NAME: 'NOT_DEFINE'
  AZURE_MODEL2_OUTPUT_CONTAINER: 'NOT_DEFINE'
  AZURE_STORAGE_WORKSPACE_CONTAINER: 'NOT_DEFINE'

steps:  
          - template: updateInstalls.yml  # Template reference

          - task: AzureCLI@2
            displayName: Get storage account name
            inputs:
              azureSubscription: '${{parameters.DEPLOYMENT_SERVICE_CONNECTION}}' 
              scriptType: 'pscore'
              scriptLocation: 'inlineScript'
              addSpnToEnvironment: true
              inlineScript: |

                 gci -recurse * | select directory, name

                 gci env: |  where Name -notmatch token | format-table -wrap
                 
                 $storageAccounts=(az resource list -o json | convertfrom-json) | where type -eq "Microsoft.Storage/storageAccounts"
                 $storageAccounts | format-table
                 $AzureStorageAccount='${{parameters.AZURE_STORAGE_ACCOUNT}}'
                 # if a specific storage account was not specified, take the storage account from the service connection
                 if ($AzureStorageAccount -eq 'NOT_DEFINE') {
                    $AzureStorageAccount = ($storageAccounts | select -First 1).name                   
                    }
                 Write-Host "##vso[task.setvariable variable=AzureStorageAccount;]$AzureStorageAccount"
                 Write-host "using storage account $AzureStorageAccount"
            
          - task: PowerShell@2
            displayName: 'Check Environment variables'
            inputs:
                targetType: 'inline'
                script: |
                  # Write your PowerShell commands here.
                      
                  gci env: | format-table -wrap
                  gci  -recurse | select fullname
          
          - task: AzureCLI@2
            displayName: Get aks cluster name
            inputs:
              azureSubscription: '${{parameters.DEPLOYMENT_SERVICE_CONNECTION}}'
              scriptType: 'pscore'
              scriptLocation: 'inlineScript'
              addSpnToEnvironment: true
              inlineScript: |
                # we arent using az aks list anymore as it's not working consistently in all environments
                $resources = az resource list -o json | convertfrom-json       
                $aksList = $resources | where 'type' -eq 'Microsoft.ContainerService/managedClusters'
                $aksList | format-table
                $aksName='$(aksName)'
                $aksResourceGroup='$(aksResourceGroup)'
                # use specific aks is named, otherwise derive from service connection permissions
                if (('set in script' -eq $aksResourceGroup) -or ('set in script' -eq $aksName))
                {
                  write-host "Setting aks from service connection environment"
                  $aksName = ($aksList | select -First 1).name
                  $aksResourceGroup= ($aksList | select -First 1).resourceGroup
                }
                Write-host "using aks $aksName resource group $aksResourceGroup "
                Write-Host "##vso[task.setvariable variable=aksName;]$aksName"
                Write-Host "##vso[task.setvariable variable=aksResourceGroup;]$aksResourceGroup"  
          
          - task: AzureCLI@2
            displayName: use kubelogin to connect
            inputs:
              azureSubscription: '${{parameters.DEPLOYMENT_SERVICE_CONNECTION}}'
              scriptType: 'bash'
              scriptLocation: 'inlineScript'
              inlineScript: |

                Write-host "using aks $(aksName) resource group $(aksResourceGroup)"
                rm -f ".kube/config" &&  az aks get-credentials --resource-group '$(aksResourceGroup)' \
                      --name '$(aksName)' \
                      --overwrite-existing 

                kubelogin convert-kubeconfig -l azurecli

                kubectl config set-context --current --namespace '${{parameters.NAMESPACE}}'

                kubectl get pods 2>&1

          - task: AzureCLI@2
            displayName: apply manifest
            inputs:
              azureSubscription: '${{parameters.DEPLOYMENT_SERVICE_CONNECTION}}'
              scriptType: 'pscore'
              powerShellErrorActionPreference: continue
              scriptLocation: 'inlineScript'
              inlineScript: |
                
                kubectl delete secret         $(jobName)-secret  2>&1 | Out-Null
                kubectl create secret generic $(jobName)-secret --from-literal=AZURE_SAS_TOKEN='${{parameters.AZURE_STORAGE_ACCOUNT_SAS_TOKEN}}'

                kubectl delete configmap      $(jobName)-config  2>&1 | Out-Null
                kubectl create configmap      $(jobName)-config --from-literal=AZURE_STORAGE_ACCOUNT=$(AzureStorageAccount) `
                                                                --from-literal=WORKSPACE=azure `
                                                                --from-literal=AZURE_MODEL2_OUTPUT_CONTAINER=${{parameters.AZURE_MODEL2_OUTPUT_CONTAINER}} `
                                                                --from-literal=AZURE_STORAGE_WORKSPACE_CONTAINER=${{parameters.AZURE_STORAGE_WORKSPACE_CONTAINER}} `
                                                                --from-literal=READ="azure"  `
                                                                --from-literal=WRITE="azure"  `
                                                                --from-literal=BYPASS_DOTENV="1"

                # because this is a job, not a deployment, we do not have to do the "kubectl rollout restart"
                kubectl delete cronjob $(jobName)-cronjob    2>&1 | Out-Null
                cd '$(Pipeline.Workspace)'
                $jobManifestFile = ((gci -recurse $(jobManifest)) | select -first 1)
                kubectl apply -f  $jobManifestFile.fullname

          - task: AzureCLI@2
            displayName: 'Run once and Check job progress'
            inputs:
              azureSubscription: '${{parameters.DEPLOYMENT_SERVICE_CONNECTION}}'
              scriptType: 'pscore'
              powerShellErrorActionPreference: continue
              scriptLocation: 'inlineScript'
              inlineScript: |

                echo "== Run once job =="
                $timestamp= Get-Date -Format "yyyyMMddHHmm"
                $jobname = "$(jobName)-runonce-rel-$(Build.BuildNumber)-time-$Timestamp"

                kubectl create job --from=cronjob/$(jobName)-cronjob $jobname

                # this step replaces the "wait" function in the kubernetes task, which is not
                # working properly for extremely long runnign jobs..
                write-host "kubernetes resources available:"

                #for the logs
                kubectl get pods  2>&1 
                kubectl get jobs  2>&1
                kubectl get services  2>&1

                # there may be more than one job, scope to just the one this pipeline cares about
                
                  try {
                      $podSpec = kubectl get pods --selector=job-name=$jobname  -o json 2> $null | convertfrom-json
                      $podname = $podspec.items[0].metadata.name
                  }  
                      catch {}
                
                $completionTime=$null
                $logs=""
                $Failed = $false
                while (-not $failed -and $podName -and -not $completionTime) {
                
                  $jobSpec = kubectl get jobs  --selector=job-name=$jobname  -o json 2> $null | convertfrom-json
                  $completionTime = $jobspec.items[0].status.completionTime
                  $success=$jobspec.items[0].status.succeeded
                  $initFailed= ($pods.items | select -First 1).status.initContainerStatuses | where .state.reason -match 'Error' 
                  $failed= (($job.status.conditions.type -eq 'Failed') -or $initFailed)

                  $previouslogs = $logs

# daily-extract-azure-pipeline.yaml

# Starter pipeline
# Start with a minimal pipeline that you can customize to build and deploy your code.
# Add steps that build, run tests, deploy, and more:
# https://aka.ms/yaml

trigger: none
pool: "msalowgts-devopvmss"
pr: none
schedules:
- cron: "0 12 * * *"
  displayName: Daily at 7am (UTC)
  branches:
    include:
    - master
  always: true
variables:
- group: 'Azure Variables'
stages:
- stage: prod
  jobs:
  - job: run_extract
    steps:
    - task: AzureCLI@2
      name: "Run_Extract"
      inputs:
        azureSubscription: 'cfo-apps-revpred-deploy-prod-AKSLOWPRODPRV002'
        scriptType: 'pscore'
        scriptLocation: 'inlineScript'
        inlineScript: |

          Install-Module -Name Az.Storage -AllowClobber -Scope CurrentUser -Force
          
          $contextFrom = New-AzStorageContext -StorageAccountName '$(AzureStorageAccountProd)' -SasToken '$(AzureSASTokenProd)'
          $contextTo =  New-AzStorageContext -StorageAccountName '$(AzureStorageAccountExtract)' -SasToken '$(AzureSASTokenExtract)'

          $blobContainers=@()
          Get-AzStorageContainer -context $contextFrom -verbose
          $original=Get-AzStorageContainer -context $contextFrom 
          $original |  % { $blobContainers += Get-AzStorageBlob -Container $_.Name -Context $contextFrom }

          [DateTime] $LastModifiedSince = [DateTime]::Now.AddDays(-1)

          $modifiedBlobs = $blobContainers | ?{ ($_.LastModified -gt  "$LastModifiedSince")}
          $modifiedContainers = $blobContainers | select -expand BlobClient  | select -expand BlobContainerName -unique
          write-host "modified containers"
          $modifiedContainers | Out-String | Write-host
          write-host "modified blobs"
          $modifiedBlobs | Out-String | Write-host
          
          # create containers needed for the blob copy
          $modifiedContainers | % { 
            If (-not (Get-AzStorageContainer -name $_ -Context $contextTo )) { 
              write-host New-AzStorageContainer -Name "$_"  -Context $contextTo.StorageAccountName 
              if (-not $dryrun) {
                New-AzStorageContainer -Name "$_"  -Context $contextTo 
                }
              }
            }

          $modifiedBlobs | % {
              # copy the blob
              Write-host Start-AzStorageBlobCopy `
                  -SrcContainer $_.BlobClient.BlobContainerName `
                  -SrcBlob $_.Name `
                  -DestContainer $_.BlobClient.BlobContainerName `
                  -DestBlob $_.Name  `
                  -Context $contextFrom.StorageAccountName `
                  -DestContext $contextTo.StorageAccountName `
                  -Force
              if (-not $dryrun) {
                  Start-AzStorageBlobCopy `
                    -SrcContainer $_.BlobClient.BlobContainerName `
                    -SrcBlob $_.Name `
                    -DestContainer $_.BlobClient.BlobContainerName `
                    -DestBlob $_.Name  `
                    -Context $contextFrom `
                    -DestContext $contextTo `
                    -Force
                    }
                  }
          # clean any files greater than the specified number of days old from extract
          $daysToKeep = "$(daysToKeep)"
          if ($daysToKeep -ne "") {
            $blobs=@()
            Get-AzStorageContainer -context $contextTo | % { 
                    $blobs += Get-AzStorageBlob -Container $_.Name -Context $contextTo
                    }
            write-host "found files on $contextTo.StorageAccountName"
            $blobs | select -expand BlobClient | Out-String | Write-host
            $old = $blobs | ?{ ($_.LastModified -lt  $([DateTime]::Now.AddDays(-$daysToKeep))) } 
            write-host "Removing files more than $(daysToKeep) days old"
            $old | select -expand BlobClient | Out-String | Write-host
            if (-not $dryrun) {
              $old | Remove-AzStorageBlob -context $contextTo
              }
            }

Function Get-AzBlobList {
        [cmdletbinding()]
        Param (
        $context 
        )


        $contents = Get-AzStorageContainer -context $context
        $allContents = @()
        foreach ($storageContainer in $contents) {
                $blobContents = Get-AzStorageBlob -context $context -container $storageContainer.name
                foreach ($blob in $blobContents) {
                        $allContents += [PSCustomObject] @{
                                AccountName =   $StorageAccountName
                                ContainerName = $storageContainer.name
                                BlobName =          $blob.Name             
                                BlobType =      $blob.BlobType
                                Length =        $blob.Length  
                                ContentType =   $blob.ContentType              
                                LastModified =  $blob.LastModified    
                                AccessTier =    $blob.AccessTier
                                SnapshotTime =  $blob.SnapshotTime             
                                IsDeleted =     $blob.IsDeleted
                                VersionId =     $blob.VersionId
                        }
                }
        }

        return $allContents
}

Function Check-StorageContainerPresenceWithoutActualTest {
    [cmdletbinding()]
    Param (
        $list,
        $context, 
        [string] $container,
        [string] $blob,
        [DateTime] $LastModifiedSince
    )
    
    $nameAndDateConstrainedList = $list | Where-Object { ($_.ContainerName -eq $container) -and ($_.blobName -like $blob) -and ($_.LastModified -gt $LastModifiedSince) }
    return ($nameAndDateConstrainedList.count -gt 0)
}


Function Test-StorageContainerPresence {
        [cmdletbinding()]
        Param (
        $list,
        $context, 
        [string] $container,
        [string] $blob,
        [string] $desiredColumns = 'yes',
        [DateTime] $LastModifiedSince
        )

        $nameConstrainedList = @($list  | ?{ ( $_.ContainerName -eq  "$container") -and ( $_.blobName -like  "$blob" ) } )
    $lastConstrainedEntry = $nameConstrainedList | Sort-Object LastModified -Descending | select-object -first 1

        Describe "Container $container has a blob $blob which is not empty and has been updated" {
                It "Container $container Blob $blob exists" {
                        $nameConstrainedList.Count | should BeGreaterThan 0 
                }
                
                It "Container $container Blob $blob is of length > 0" {
                        $lastConstrainedEntry.Length | should BeGreaterThan 0  
                }
                
                It "Container $container Blob $blob is not deleted" {
                        $lastConstrainedEntry.IsDeleted | should Be False  
                }
        
                if ($LastModifiedSince) {

                        $hoursSince = [math]::Round(([DateTime]::Now - $LastModifiedSince ).TotalHours)     
            
                        It "Container $container Blob $blob has been updated in the past $hoursSince hours" {
                                        $lastConstrainedEntry.LastModified | should  BeGreaterThan $LastModifiedSince  
            }
                }



        if (("$blob" -match ".csv") -and ($desiredColumns) -and ($lastConstrainedEntry)) {

                        $tmp = New-TemporaryFile

                        $testBlobContent = Get-AzStorageBlobContent -context $context -container $lastConstrainedEntry.ContainerName -blob $lastConstrainedEntry.blobName -Destination $tmp.name -force

                        $truncatedtmp = New-TemporaryFile

                        # csv files can be safely truncated, and this REALLY speeds up processing
                        # we run this twice so column names don't get trimmed for "not having contents" 
                        $content =  @(Get-Content $tmp.name -first 1)
                        $content += @(Get-Content $tmp.name -first 1)
                        $content | Set-Content $truncatedtmp.name

                        $blobcsv = import-csv $truncatedtmp.name

                        if ($blobcsv) {
                                $blobCsvColumns = ($blobcsv | Get-Member -MemberType Properties | Select-Object -ExpandProperty Name | Select-Object -Unique | %{"`'$_`'"})
                                $BlobCsvColumnNames = [string]::Join(", ", $blobCsvColumns)
                        }

                        $lastConstrainedEntry | Add-Member -NotePropertyName CsvColumnNames -NotePropertyValue $BlobCsvColumnNames -force

                        It "Container $container blob $($lastConstrainedEntry.blobName) has column count $($blobCsvColumns.count) should be greater than 0" {
                                $($blobCsvColumns.count) | should BeGreaterThan 0  
                        }

                        It "Container $container blob $($lastConstrainedEntry.blobName) has column names: { $($blobCsvColumnNames) } should not be empty" {
                                $($blobCsvColumnNames) | should not BeNullOrEmpty  
                        }
                        
                        del $tmp.Name
                        del $truncatedtmp.name
                }

        }
    
    write-verbose "$lastConstrainedEntry"
    
        return $lastConstrainedEntry
}

🔎 High-Level Summary

These YAML and shell script files together implement an Azure DevOps CI/CD pipeline for a Revenue Prediction system that:

Prepares environments

Installs required tools (zip, helm, kubelogin, Azure PowerShell modules).

Cleans up Docker images on build agents when needed.

Creates training jobs for machine learning models (M1, M2, M3, M4)

Each training pipeline (training-pipeline-azure.yaml, m3-training-pipeline-azure.yaml, m4-training-pipeline-azure.yaml)

Sets a timestamp to tag training runs.

Runs a Dev AKS job that generates training data and stores results in Azure Blob Storage.

If successful, conditionally runs a Prod AKS job for official production training.

Uses deployTrainJobManifest.yml (not shown but referenced) to deploy a Kubernetes Job manifest to AKS.

Handles artifacts and configuration

Passes variables (like timestamp, storage account, SAS tokens, job manifest) between stages.

Publishes and downloads artifacts (scratchpad timestamp files, training results).

Automates scheduled runs

Quarterly cron schedules trigger retraining on master automatically.

Pipelines also trigger from upstream build pipelines (e.g., model training images being rebuilt).

Generates Excel outputs via APIs

Scripts (dev.create-excel-files.sh, prod.create-excel-files.sh)

Call internal APIs (/api/excel) with different business units (BUs) and sub-units.

Automates the creation of Excel-based reports in dev and prod environments.

Deployment & Job execution flow

YAML pipelines orchestrate job execution on Kubernetes (AKS).

They mount secrets and parameters (service connections, SAS tokens, namespaces).

Jobs run the actual training containers (revenue-prediction-mX-train) to process data.

📌 In Plain Words

This is a full CI/CD system for ML model training and report generation.

Developers commit code → triggers pipelines.

Pipelines build images → training pipelines pick them up.

Training pipelines:

Generate a timestamp,

Launch training jobs on Dev AKS,

If successful, repeat on Prod AKS.

Artifacts and results are stored in Azure Blob Storage.

Excel report scripts call APIs to prepare business-unit–specific reports.

Utility pipelines handle dependencies (install tools, clean images).

Effectively, it ensures that models are retrained quarterly (or on demand), stored properly, and aligned between dev/prod, while Excel reports are generated for different business units automatically.
